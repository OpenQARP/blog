{{- /* Horizontal bars from "label | value" lines, scaled from zero to the largest value. With `highlight`, that row takes the accent and the others go neutral. {{< bars title="…" note="…" unit=" ms" highlight="OpenQARP" >}} */ -}} {{- $rows := slice -}} {{- $max := 0.0 -}} {{- range split (trim .Inner "\n") "\n" -}} {{- $cells := split . "|" -}} {{- if eq (len $cells) 2 -}} {{- $label := trim (index $cells 0) " " -}} {{- $value := trim (index $cells 1) " " -}} {{- $rows = $rows | append (dict "label" $label "value" $value) -}} {{- $max = math.Max $max (float (replace $value "," "")) -}} {{- else if trim . " " -}} {{- errorf "bars in %s: expected \"label | value\", got %q" $.Page.Path . -}} {{- end -}} {{- end -}} {{- $highlight := .Get "highlight" -}}
